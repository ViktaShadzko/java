HashMap — основные операции

Map<String, Integer> scores = new HashMap<>();
// Добавление
scores.put("Иван",  90);
scores.put("Мария", 85);
scores.put("Пётр",  92);
scores.put("Иван",  95); // перезапишет!
// Получение
scores.get("Иван");                   // 95
scores.get("Нет");                    // null
scores.getOrDefault("Нет", 0);        // 0
// Проверка
scores.containsKey("Мария");          // true
scores.containsValue(85);             // true
scores.size();                        // 3
scores.isEmpty();                     // false
// Удаление
scores.remove("Пётр");
scores.remove("Мария", 85); // если совпадает
// Итерация — лучший способ
scores.forEach((name, score) ->
    System.out.println(name+" → "+score));
// Через entrySet
for (Map.Entry<String, Integer> e
        : scores.entrySet()) {
    System.out.println(
        e.getKey() + " → " + e.getValue());
}
// Только ключи / значения
Set<String> keys = scores.keySet();
Collection<Integer> vals = scores.values();
// Stream
scores.entrySet().stream()
    .filter(e -> e.getValue() >= 90)
    .sorted(Map.Entry
        .comparingByValue(
            Comparator.reverseOrder()))
    .forEach(e ->
        System.out.println(e.getKey()));

Java 8+ методы и TreeMap

// computeIfAbsent — создать если нет
Map<String, List<String>> groups =
    new HashMap<>();
// Старый способ:
if (!groups.containsKey("IT"))
    groups.put("IT", new ArrayList<>());
groups.get("IT").add("Иван");
// Java 8+ — в одну строку:
groups.computeIfAbsent("IT",
    k -> new ArrayList<>()).add("Иван");
// merge — подсчёт слов
String[] words = {"яблоко","банан","яблоко"};
Map<String, Integer> count = new HashMap<>();
for (String w : words)
    count.merge(w, 1, Integer::sum);
// {яблоко=2, банан=1}
// TreeMap — всегда отсортирован
TreeMap<Integer, String> tree =
    new TreeMap<>();
tree.put(90, "Иван");
tree.put(85, "Мария");
tree.put(92, "Пётр");
System.out.println(tree.firstKey()); // 85
System.out.println(tree.lastKey());  // 92
// Диапазон от 85 до 92 включительно
tree.subMap(85,true, 92,true).forEach(
    (s,n) -> System.out.println(n+": "+s));
// LinkedHashMap — LRU Cache
class LRUCache<K,V>
        extends LinkedHashMap<K,V> {
    private final int max;
    LRUCache(int max) {
        super(max, 0.75f, true);
        this.max = max;
    }
    @Override protected boolean
        removeEldestEntry(
              Map.Entry<K,V> e) {
        return size() > max;
    }
}

📌 Итоги

  • HashMap — O(1), нет порядка. Выбор по умолчанию.
  • TreeMap — O(log n), отсортирован, поддерживает диапазоны.
  • LinkedHashMap — порядок вставки. Основа LRU Cache.
  • computeIfAbsent, merge, getOrDefault — Java 8+.
  • Ключи должны корректно реализовывать equals и hashCode.
