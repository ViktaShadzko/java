Locale и ResourceBundle

// ── Locale ─────────────────────────────
Locale ru = new Locale("ru", "RU");
Locale us = Locale.US;
Locale de = Locale.GERMANY;
System.out.println(ru.getLanguage());  // ru
System.out.println(ru.getCountry());   // RU
System.out.println(ru.getDisplayName());// русский(Россия)
// ── ResourceBundle (.properties файлы)
// messages.properties (по умолчанию/English):
// greeting=Hello, {0}!
// books.found=Found {0} books
// messages_ru.properties:
// greeting=Привет, {0}!
// books.found=Найдено книг: {0}
ResourceBundle bundle =
    ResourceBundle.getBundle("messages",
        new Locale("ru","RU"));
System.out.println(bundle.getString("greeting"));
// Привет, {0}!
// MessageFormat для подстановки
String tpl = bundle.getString("greeting");
String msg = MessageFormat.format(tpl, "Иван");
System.out.println(msg); // Привет, Иван!
// ── Числа и валюты ──────────────────
double price = 1234567.89;
NumberFormat nfRu = NumberFormat
    .getNumberInstance(ru);
System.out.println(nfRu.format(price));
// 1 234 567,89
NumberFormat cfRu = NumberFormat
    .getCurrencyInstance(ru);
System.out.println(cfRu.format(price));
// 1 234 567,89 ₽
NumberFormat cfUs = NumberFormat
    .getCurrencyInstance(Locale.US);
System.out.println(cfUs.format(price));
// $1,234,567.89

java.time API

// ── LocalDate — только дата ─────────
LocalDate today    = LocalDate.now();
LocalDate birthday = LocalDate.of(2000, 3, 15);
LocalDate tomorrow = today.plusDays(1);
LocalDate nextYear = today.plusYears(1);
System.out.println(today);               // 2026-03-04
System.out.println(today.getDayOfWeek()); // WEDNESDAY
System.out.println(today.isAfter(birthday)); // true
// ── LocalTime — только время ─────────
LocalTime now  = LocalTime.now();
LocalTime start = LocalTime.of(9, 0);
LocalTime noon  = LocalTime.NOON;
// ── LocalDateTime — дата + время ─────
LocalDateTime dt =
    LocalDateTime.now();
LocalDateTime meeting =
    LocalDateTime.of(2026, 3, 10, 14, 30);
// ── Period и Duration ─────────────────
Period age = Period.between(birthday, today);
System.out.printf("Возраст: %d лет %d мес.%n",
    age.getYears(), age.getMonths());
Duration work = Duration.between(
    LocalDateTime.of(2026,3,4, 9,0),
    LocalDateTime.of(2026,3,4,17,30));
System.out.println(work.toHours());   // 8
System.out.println(work.toMinutes()); // 510
// ── DateTimeFormatter ────────────────
DateTimeFormatter ruDate =
    DateTimeFormatter.ofPattern("dd.MM.yyyy");
DateTimeFormatter withDay =
    DateTimeFormatter.ofPattern(
        "EEEE, dd MMMM yyyy",
        new Locale("ru"));
LocalDate d = LocalDate.of(2026, 3, 4);
System.out.println(d.format(ruDate));
// 04.03.2026
System.out.println(d.format(withDay));
// среда, 04 марта 2026
// Парсинг
LocalDate parsed =
    LocalDate.parse("04.03.2026", ruDate);
// ── Часовые пояса ─────────────────────
ZoneId moscow = ZoneId.of("Europe/Moscow");
ZoneId ny     = ZoneId.of("America/New_York");
ZonedDateTime nowMoscow =
    ZonedDateTime.now(moscow);
ZonedDateTime nowNY =
    nowMoscow.withZoneSameInstant(ny);
System.out.println("Москва: "+nowMoscow.toLocalTime());
System.out.println("Нью-Йорк: "+nowNY.toLocalTime());

📌 Итоги

  • Locale — язык + регион. Locale.US, new Locale("ru","RU")
  • ResourceBundle — строки в .properties по локали
  • NumberFormat — числа, валюты, проценты по локали
  • LocalDate, LocalTime, LocalDateTime — иммутабельны, thread-safe
  • Period — разница в датах (год/мес/день)
  • Duration — разница во времени (часы/мин/сек)
  • ZonedDateTime — работа с часовыми поясами
