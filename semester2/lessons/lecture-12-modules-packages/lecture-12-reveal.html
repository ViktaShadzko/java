Пакеты и видимость

// ── Объявление пакета — 1-я строка ────
package com.company.project.service;
// Путь: src/com/company/project/service/UserService.java
// ── import ─────────────────────────────
import java.util.List;
import java.util.ArrayList;
import java.util.*;                  // wildcard
import static java.lang.Math.PI;     // статический
import static java.lang.Math.sqrt;
// ── Соглашение об именах ───────────────
// com.company.project.layer
// com.bank.payments.service.TransactionService
// org.university.course.repository.StudentRepository
// ── Модификаторы доступа ───────────────
public    class A {}  // виден везде
          class B {}  // package-private: только пакет
protected class C {}  // нет смысла для класса верх. уровня
// ── package-private на практике ───────
// com.example.service.OrderService
package com.example.service;
public class OrderService {
    public  void createOrder()   { }  // все
    void    validateOrder()      { }  // только service пакет
    private void calculateTax()  { }  // только этот класс
}
// com.example.service.PaymentService
package com.example.service;
public class PaymentService {
    public void process(OrderService os) {
        os.createOrder();   // ✅ public
        os.validateOrder(); // ✅ тот же пакет!
    }
}
// com.example.controller.OrderController
package com.example.controller;
public class OrderController {
    public void handle(OrderService os) {
        os.createOrder();   // ✅ public
        // os.validateOrder(); // ❌ другой пакет!
    }
}

Модульная система JPMS (Java 9+)

// ── module-info.java ───────────────────
module com.example.app {
    // Зависимости
    requires java.sql;
    requires com.google.gson;
    requires transitive java.logging; // транзитивная
    // Экспорт — что мы открываем
    exports com.example.app.service;  // всем
    exports com.example.app.model;
    exports com.example.app.api
        to com.example.client;        // только клиенту
    // Для рефлексии (Hibernate, Jackson)
    opens com.example.app.model;
    opens com.example.app.internal
        to com.google.gson;
}

📌 Итоги

  • Пакет — пространство имён + управление доступом
  • package-private — виден только в том же пакете
  • Соглашение: обратный домен com.company.project.layer
  • Модули JPMS — явные зависимости и экспорт (Java 9+)
  • requires — зависимости, exports — что открыть
  • opens — разрешить рефлексию (нужно Hibernate/Jackson)
