Collections Framework — дерево

Iterable
└── Collection
    ├── List      — порядок + дубликаты
    │   ├── ArrayList  ← get O(1)
    │   └── LinkedList ← add/remove O(1)
    ├── Set       — уникальные элементы
    │   ├── HashSet       — O(1), нет порядка
    │   ├── LinkedHashSet — порядок вставки
    │   └── TreeSet       — отсортирован
    └── Queue     — очередь
        ├── PriorityQueue — по приоритету
        └── Deque
            └── ArrayDeque ← стек/очередь
// List — основные операции
List<String> list = new ArrayList<>();
list.add("Иван");
list.add(0, "Анна");   // по индексу
list.get(0);            // O(1) для ArrayList
list.set(0, "Новое");
list.remove(0);
list.contains("Иван");
list.size();
// Неизменяемые списки
List<String> fixed = List.of("a","b","c");
// Сортировка
list.sort(Comparator.naturalOrder());
list.sort(Comparator.comparingInt(
    String::length));

Set и важность equals/hashCode

// HashSet — O(1), нет порядка
Set<String> set = new HashSet<>();
set.add("банан"); set.add("яблоко");
set.add("банан"); // дубликат!
System.out.println(set.size()); // 2
// Множества
Set<Integer> a = new HashSet<>(
    List.of(1,2,3,4,5));
Set<Integer> b = new HashSet<>(
    List.of(3,4,5,6,7));
Set<Integer> intersection = new HashSet<>(a);
intersection.retainAll(b); // [3,4,5]
Set<Integer> union = new HashSet<>(a);
union.addAll(b);           // [1,2,3,4,5,6,7]
// ❌ Без equals/hashCode
class BadPoint { int x, y; }
Set<BadPoint> s = new HashSet<>();
s.add(new BadPoint(1,2));
s.add(new BadPoint(1,2)); // дубликат!
System.out.println(s.size()); // 2 ❌
// ✅ С equals/hashCode
class Point {
    int x, y;
    @Override public boolean equals(Object o){
        if (!(o instanceof Point)) return false;
        Point p = (Point) o;
        return x==p.x && y==p.y;
    }
    @Override public int hashCode(){
        return Objects.hash(x, y);
    }
}
Set<Point> sp = new HashSet<>();
sp.add(new Point(1,2));
sp.add(new Point(1,2));
System.out.println(sp.size()); // 1 ✅

Queue, Deque, PriorityQueue

// Queue — FIFO
Queue<String> q = new LinkedList<>();
q.offer("первый");
q.offer("второй");
System.out.println(q.peek()); // первый
System.out.println(q.poll()); // первый
// Deque — стек LIFO (лучше чем Stack!)
Deque<String> stack = new ArrayDeque<>();
stack.push("A"); stack.push("B");
System.out.println(stack.pop()); // B
// PriorityQueue — min-heap
PriorityQueue<Integer> pq =
    new PriorityQueue<>();
pq.offer(5); pq.offer(1); pq.offer(3);
while (!pq.isEmpty())
    System.out.print(pq.poll()+" "); // 1 3 5
// Шпаргалка выбора:
// доступ по индексу  → ArrayList
// вставка в начало   → LinkedList / ArrayDeque
// уникальные         → HashSet
// уникальные+сорт.   → TreeSet
// приоритет          → PriorityQueue
// стек/очередь       → ArrayDeque

📌 Итоги

  • ArrayList — быстрый get, выбор по умолчанию
  • LinkedList — быстрые insert/delete в начале
  • HashSet — уникальные, O(1)
  • TreeSet — уникальные + отсортированные
  • ArrayDeque — лучший стек и очередь
  • Всегда переопределяй equals + hashCode вместе!
