Типичные ошибки

// ❌ Ошибка 1: NullPointerException
String s = null;
s.length(); // NPE!
// ✅ Решение:
if (s != null) s.length();
Optional.ofNullable(s).map(String::length);
// ❌ Ошибка 2: Изменение коллекции в цикле
for (String item : list) {
    if (item.equals("del"))
        list.remove(item); // ConcurrentModificationException!
}
// ✅ Решение:
list.removeIf(item -> item.equals("del"));
// ❌ Ошибка 3: Нет equals/hashCode
class BadPoint { int x, y; }
Set<BadPoint> set = new HashSet<>();
set.add(new BadPoint()); set.add(new BadPoint());
System.out.println(set.size()); // 2 (должно быть 1!)
// ❌ Ошибка 4: Широкий catch
try { riskyOperation(); }
catch (Exception e) { // всё подряд!
    System.out.println("Что-то пошло не так");
} // теряем информацию об ошибке!
// ✅ Конкретный тип + логирование
catch (FileNotFoundException e) {
    log.error("Файл не найден", e);
}
// ❌ Ошибка 5: Стрим использован дважды
Stream<String> s = list.stream().filter(...);
s.forEach(System.out::println); // OK
s.count(); // ❌ IllegalStateException!
// Стрим = ОДНОРАЗОВЫЙ!

Мини-тест и что дальше

// Вопрос 1: Что выведет?
String s1 = "hello";
String s2 = "hello";
String s3 = new String("hello");
System.out.println(s1 == s2);      // ?
System.out.println(s1 == s3);      // ?
System.out.println(s1.equals(s3)); // ?
// Вопрос 2: Что выведет?
List<Integer> nums = List.of(1,2,3,4,5);
int result = nums.stream()
    .filter(n -> n % 2 == 0)
    .mapToInt(Integer::intValue)
    .sum();
System.out.println(result); // ?
// Вопрос 3: Где ошибка?
class Stack<T> {
    private List<T> items = new ArrayList<>();
    public void push(T item) { items.add(item); }
    public T pop() {
        return items.remove(items.size()); // ← ?
    }
}
// Ответы: true, false, true | 6 | items.size()-1
