Лямбды и java.util.function

// Синтаксис лямбд
Runnable r  = () -> System.out.println("!");
Consumer<String> p = s -> System.out.println(s);
Comparator<Integer> c = (a, b) -> a - b;
Function<Integer,Integer> fac = n -> {
    int res = 1;
    for (int i=2; i<=n; i++) res *= i;
    return res;
};
System.out.println(fac.apply(5)); // 120
// ── java.util.function ─────────────────
// Predicate<T> — T → boolean
Predicate<String> isLong = s -> s.length()>5;
Predicate<String> startsA = s -> s.startsWith("А");
Predicate<String> both = isLong.and(startsA);
// Function<T,R> — T → R
Function<String,Integer> len = String::length;
Function<Integer,String> str = Object::toString;
Function<String,String> comp =
    len.andThen(str); // String→Integer→String
// Consumer<T> — T → void
Consumer<String> log  = s -> System.out.println("[LOG] "+s);
Consumer<String> save = s -> System.out.println("[SAVE] "+s);
Consumer<String> both2 = log.andThen(save);
// Supplier<T> — () → T
Supplier<List<String>> factory = ArrayList::new;
List<String> list = factory.get();
// ── Method References ───────────────────
// 1. Статический: Class::staticMethod
Function<String,Integer> parse = Integer::parseInt;
// 2. Конкретного объекта: obj::method
String prefix = "Hello, ";
Function<String,String> greet = prefix::concat;
// 3. Произвольного: Class::instanceMethod
Function<String,String>  up  = String::toUpperCase;
Predicate<String>        emp = String::isEmpty;
// 4. Конструктор: Class::new
Function<String,StringBuilder> sb =
    StringBuilder::new;
// Пример
List<String> names = List.of("  Иван  ", "  Мария  ");
names.stream()
    .map(String::trim)
    .map(String::toUpperCase)
    .forEach(System.out::println); // ИВАН МАРИЯ

Comparable, Comparator, Optional

// Comparable — естественный порядок
public class Student implements Comparable<Student> {
    private String name;
    private double gpa;
    // ...
    @Override
    public int compareTo(Student other) {
        return Double.compare(other.gpa, this.gpa);
    }
}
Collections.sort(students); // использует compareTo
// Comparator — внешний порядок
Comparator<Student> byGpaDesc = Comparator
    .comparingDouble(Student::getGpa)
    .reversed()
    .thenComparing(Student::getName);
students.sort(byGpaDesc);
// Optional — явная обёртка для nullable
Optional<String> present = Optional.of("Привет");
Optional<String> empty   = Optional.empty();
present.isPresent();              // true
present.get();                    // Привет
empty.orElse("дефолт");           // дефолт
empty.orElseGet(() -> compute());
empty.orElseThrow(
    () -> new IllegalStateException());
// Трансформации
present.map(String::length);      // Optional[6]
present.filter(s -> s.length()>3);// Optional[Привет]
// Цепочка (избегаем NPE)
Optional.ofNullable(user)
    .map(User::getAddress)
    .map(Address::getCity)
    .orElse("Город неизвестен");

📌 Итоги

  • Лямбды — краткая реализация функциональных интерфейсов
  • Predicate, Function, Consumer, Supplier — java.util.function
  • Method References — ещё короче лямбд
  • Comparable — внутренний порядок. Comparator — внешний.
  • Optional — явная обёртка, замена null
  • Лямбды захватывают только effectively final переменные
