Stream — декларативная обработка

// ── Без Stream (императивно) ───────────
List<String> result = new ArrayList<>();
for (String name : names) {
    if (name.length() > 4) {
        result.add(name.toUpperCase());
    }
}
Collections.sort(result);
// ── С Stream (декларативно) ────────────
List<String> result = names.stream()
    .filter(name -> name.length() > 4)
    .map(String::toUpperCase)
    .sorted()
    .collect(Collectors.toList());
// Создание Stream
Stream<String> s1 = list.stream();
Stream<String> s2 = Stream.of("a","b","c");
IntStream      s3 = IntStream.range(1, 6);
// Из генераторов
Stream.iterate(0, n -> n+2)  // 0,2,4,6...
      .limit(5)
      .forEach(System.out::println);
// ── Промежуточные операции ──────────────
// filter, map, flatMap, sorted, distinct, 
// limit, skip, peek
List<List<Integer>> nested = List.of(
    List.of(1,2), List.of(3,4), List.of(5));
// flatMap — разворачивает вложенные
nested.stream()
    .flatMap(Collection::stream)
    .forEach(System.out::print); // 1 2 3 4 5
// ── Терминальные операции ───────────────
long count = stream.count();
Optional<T> first = stream.findFirst();
boolean any = stream.anyMatch(pred);
boolean all = stream.allMatch(pred);
// reduce — свёртка
int product = IntStream.rangeClosed(1,5)
    .reduce(1, (a,b) -> a*b); // 120

Collectors и ленивость

List<Student> students = ...;
// Группировка
Map<String, List<Student>> byFaculty =
    students.stream()
        .collect(Collectors.groupingBy(
            Student::getFaculty));
// Группировка + счётчик
Map<String, Long> countByFac =
    students.stream()
        .collect(Collectors.groupingBy(
            Student::getFaculty,
            Collectors.counting()));
// Разбивка по условию
Map<Boolean, List<Student>> partition =
    students.stream()
        .collect(Collectors.partitioningBy(
            s -> s.getGpa() >= 8.5));
// Статистика
DoubleSummaryStatistics stats =
    students.stream()
        .collect(Collectors.summarizingDouble(
            Student::getGpa));
System.out.println(stats.getAverage());
System.out.println(stats.getMax());
// IntStream — без boxing!
int sum = IntStream.rangeClosed(1,100)
    .sum(); // 5050 (формула Гаусса!)
IntSummaryStatistics s =
    IntStream.of(3,1,4,1,5,9)
        .summaryStatistics();
// Ленивость стримов — важно!
Stream<Integer> pipeline =
    Stream.of(1,2,3,4,5)
        .filter(n -> {
            System.out.println("filter:"+n);
            return n % 2 == 0;
        })
        .map(n -> {
            System.out.println("map:"+n);
            return n * 10;
        });
// Ничего не выведено! Стрим ленивый.
List<Integer> result = pipeline
    .collect(Collectors.toList());
// Теперь: filter:1 filter:2 map:2 filter:3
// filter:4 map:4 filter:5 → [20, 40]

📌 Итоги

  • Stream — не хранит данные, обрабатывает. Ленивый.
  • Промежуточные: filter, map, flatMap, sorted, distinct
  • Терминальные: collect, count, reduce, findFirst, anyMatch
  • Collectors.groupingBy, partitioningBy, joining
  • IntStream — без boxing, есть range, sum, average
  • Стрим = одноразовый. После терминальной операции закрыт!
