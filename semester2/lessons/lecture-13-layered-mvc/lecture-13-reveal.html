Repository, Service, Controller

// ── Repository — только CRUD ──────────
public interface BookRepository {
    Book save(Book book);
    Optional<Book> findById(Long id);
    List<Book> findAll();
    List<Book> findByAuthor(String author);
    void delete(Long id);
}
// ── Service — бизнес-логика ───────────
public class BookService {
    // Зависимость через ИНТЕРФЕЙС!
    private final BookRepository repo;
    public BookService(BookRepository r) {
        this.repo = Objects.requireNonNull(r);
    }
    public Book addBook(String title,
                        double price) {
        // Валидация — это бизнес-правило!
        if (title == null || title.isBlank())
            throw new ValidationException(
                "Название обязательно");
        if (price < 0)
            throw new ValidationException(
                "Цена не может быть < 0");
        return repo.save(new Book(title, price));
    }
    public void borrowBook(Long bookId,
                           String reader) {
        Book book = repo.findById(bookId)
            .orElseThrow(() ->
                new BookNotFoundException(bookId));
        if (!book.isAvailable())
            throw new BookNotAvailableException();
        book.setAvailable(false);
        repo.save(book);
    }
}
// ── Controller — тонкий слой ──────────
public class BookController {
    private final BookService service;
    public BookController(BookService s) {
        this.service = s;
    }
    public void addBook(String title,
                        double price) {
        try {
            Book b = service.addBook(title,price);
            System.out.println("✅ "+b);
        } catch (ValidationException e) {
            System.out.println("❌ "+e.getMessage());
        }
    }
}
// ── Main — сборка (Composition Root) ──
public class Main {
    public static void main(String[] args) {
        // 1. Создаём инфраструктуру
        BookRepository repo =
            new InMemoryBookRepository();
        // 2. Внедряем в сервис
        BookService service = new BookService(repo);
        // 3. Внедряем в контроллер
        BookController ctrl =
            new BookController(service);
        ctrl.addBook("Java", 450.0);
        ctrl.addBook("", 0.0); // ❌ валидация
    }
}

SOLID кратко

ПринципРасшифровкаСуть
SSingle ResponsibilityОдин класс — одна ответственность
OOpen/ClosedОткрыт для расширения, закрыт для изменения
LLiskov SubstitutionНаследник заменяем родителем
IInterface SegregationМного малых интерфейсов лучше одного большого
DDependency InversionЗависеть от абстракций, не от реализаций
 
// ❌ D — зависимость от реализации
public class BookService {
    private InMemoryBookRepository repo =
        new InMemoryBookRepository();
    // нельзя поменять без изменения класса!
}
// ✅ D — зависимость от абстракции
public class BookService {
    private final BookRepository repo; // интерфейс!
    public BookService(BookRepository r) {
        this.repo = r; // внедряем снаружи
    }
    // теперь можно: InMemory, DB, или Mock!
}

📌 Итоги

  • Controller — тонкий: принять → делегировать → ответить
  • Service — вся бизнес-логика. Не знает об UI и БД.
  • Repository — только CRUD. Не знает о бизнес-логике.
  • DI — передавать зависимости через конструктор.
  • Зависеть от интерфейсов, не реализаций.
