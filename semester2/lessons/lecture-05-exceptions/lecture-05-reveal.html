Проверка компилятором

Checked: компилятор ЗАСТАВИТ обработать или объявить через throws.
Unchecked: компилятор молчит — наша ответственность.

try-catch-finally

public static String readFirstLine(
                         String path) {
    BufferedReader reader = null;
    try {
        // ── Код, который может бросить ──
        reader = new BufferedReader(
            new FileReader(path));
        return reader.readLine();

    } catch (FileNotFoundException e) {
        // ── Конкретный тип — первый ─────
        System.err.println(
          "Файл не найден: " + path);
        return null;

    } catch (IOException e) {
        // ── Более широкий — потом ───────
        System.err.println(
          "Ошибка чтения: " + e.getMessage());
        return null;

    } finally {
        // ── Выполняется ВСЕГДА ──────────
        if (reader != null) {
            try { reader.close(); }
            catch (IOException e) { }
        }
    }
}
// ✅ try-with-resources (Java 7+)
// Автоматически закрывает AutoCloseable
try (BufferedReader reader =
         new BufferedReader(
           new FileReader(path))) {
    return reader.readLine();
} catch (IOException e) {
    System.err.println(e.getMessage());
    return null;
}
// reader закрывается АВТОМАТИЧЕСКИ!
// Multi-catch (Java 7+)
try {
    processFile("data.csv");
} catch (FileNotFoundException |
         ParseException e) {
    // обработать оба типа одинаково
    log.error("Ошибка файла", e);
}
Правило

Ресурсы (файлы, соединения) — всегда через try-with-resources!

Свои исключения и Exception Chaining

// ── Checked исключение ────────────────
public class InsufficientFundsException
                      extends Exception {
    private final double requested;
    private final double available;

    public InsufficientFundsException(
            double requested,
            double available) {
        super(String.format(
          "Запрошено: %.2f, доступно: %.2f",
          requested, available));
        this.requested = requested;
        this.available = available;
    }

    public double getShortfall() {
        return requested - available;
    }
}

// ── Unchecked исключение ──────────────
public class UserNotFoundException
             extends RuntimeException {
    private final String userId;

    public UserNotFoundException(
                        String userId) {
        super("Пользователь не найден: "
              + userId);
        this.userId = userId;
    }
    // Конструктор с cause
    public UserNotFoundException(
            String userId, Throwable cause) {
        super("Не найден: " + userId, cause);
        this.userId = userId;
    }
}
// ── Exception Chaining ───────────────
public class UserService {
    public User findUser(String id) {
        try {
            return userRepo.findById(id);
        } catch (SQLException e) {
            // оборачиваем, сохраняя cause!
            throw new UserNotFoundException(
                id, e); // ← передаём причину
        }
    }
}

// Стек вывода покажет ОБА исключения:
// UserNotFoundException: Не найден: 123
//   Caused by: SQLException: ...

// ── Антипаттерны ─────────────────────
// ❌ Пустой catch — НИКОГДА!
try { risky(); }
catch (Exception e) { }

// ❌ Слишком широкий catch
try { connectDB(); process(); save(); }
catch (Exception e) { // всё подряд!
    System.out.println("Ошибка");
}

// ✅ Конкретный тип + логирование
catch (SQLException e) {
    log.error("Ошибка БД при id={}", id, e);
    throw new ServiceException("...", e);
}

📌 Итоги

  • Checked — компилятор заставляет обработать
  • Unchecked (RuntimeException) — программная ошибка
  • finally — всегда выполняется
  • try-with-resources — для Closeable объектов
  • Свои исключения: checked → extends Exception
  • Unchecked → extends RuntimeException
  • Exception chaining — передавай cause
  • Никогда не глотай исключения молча!
