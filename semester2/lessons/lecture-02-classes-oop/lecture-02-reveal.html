1. Класс и объект

Чертёж и конкретный экземпляр

Класс — чертёж. Объект — экземпляр

Аналогия

Класс Автомобиль — техническая документация. Объект myCar — конкретный автомобиль во дворе.

// Класс — чертёж
public class Car {
    String brand;
    String model;
    int    year;
    double speed;

    void accelerate(double amount) {
        speed += amount;
        System.out.println(brand +
          " разгоняется до " + speed);
    }
    void brake() {
        speed = 0;
        System.out.println(brand +
          " остановился");
    }
}
// Объекты — конкретные экземпляры
Car car1 = new Car();
car1.brand = "Toyota";
car1.model = "Camry";
car1.year  = 2023;

Car car2 = new Car();
car2.brand = "BMW";
car2.model = "M3";

car1.accelerate(60);  // Toyota → 60
car2.accelerate(100); // BMW → 100
car1.brake();         // Toyota → 0

System.out.println(car1.speed); // 0.0
System.out.println(car2.speed); // 100.0
// Объекты независимы!

2. Поля, методы, конструкторы

Структура класса Java

Структура класса

public class Student {
    // ── Поля (состояние) ───────────
    private String name;
    private int    age;
    private double gpa;

    // ── Конструктор ────────────────
    public Student(String name,
                   int age,
                   double gpa) {
        this.name = name;
        this.age  = age;
        this.gpa  = gpa;
    }

    // ── Методы (поведение) ─────────
    public void study() {
        gpa = Math.min(10.0, gpa + 0.1);
        System.out.println(name +
          " учится, GPA: " + gpa);
    }

    public String getSummary() {
        return name + " | " + age +
          " лет | GPA: " + gpa;
    }
}

📦 Поля

Состояние объекта. Обычно private.

🔧 Конструктор

Создаёт объект. Имя = имя класса.

⚙️ Методы

Поведение объекта.

// Создание объектов
Student s1 = new Student(
    "Иван", 20, 8.5);
Student s2 = new Student(
    "Мария", 21, 9.2);

s1.study();
System.out.println(s1.getSummary());
// Иван | 20 лет | GPA: 8.6
static vs instance

static — одно на весь класс.
Без static — у каждого объекта своё.

Перегрузка конструкторов и this()

public class User {
    private String name;
    private String email;
    private int    age;

    // Конструктор 1: только имя
    public User(String name) {
        this(name, "", 0); // ← вызов 3
    }

    // Конструктор 2: имя + email
    public User(String name,
                String email) {
        this(name, email, 0); // ← вызов 3
    }

    // Конструктор 3: всё
    public User(String name,
                String email,
                int age) {
        this.name  = name;
        this.email = email;
        this.age   = age;
    }
}
this() — вызов другого конструктора

Должен быть ПЕРВОЙ строкой конструктора. Избегает дублирования кода.

User u1 = new User("Иван");
User u2 = new User("Мария",
                   "m@mail.ru");
User u3 = new User("Пётр",
                   "p@mail.ru",
                   25);
this.field vs this()

this.name — ссылка на поле объекта
this() — вызов другого конструктора

3. Инкапсуляция

Скрываем детали — открываем интерфейс

Инкапсуляция — зачем?

// ❌ Без инкапсуляции
public class BadAccount {
    public double balance; // всё открыто!
}

BadAccount acc = new BadAccount();
acc.balance = -99999; // никто не мешает!
// ✅ С инкапсуляцией
public class BankAccount {
    private double balance; // скрыто!
    private String owner;

    public BankAccount(String owner,
                       double amount) {
        this.owner = owner;
        if (amount < 0) throw new
            IllegalArgumentException(
              "Баланс не может быть < 0");
        this.balance = amount;
    }

    public void deposit(double amount) {
        if (amount <= 0) throw new
            IllegalArgumentException(
              "Сумма должна быть > 0");
        balance += amount;
    }

    public void withdraw(double amount) {
        if (amount > balance)
            throw new IllegalStateException(
              "Недостаточно средств");
        balance -= amount;
    }

    public double getBalance() {
        return balance; // только чтение
    }
}

Правило

Поля — всегда private.
Доступ — через методы (геттеры/сеттеры).

✅ Зачем инкапсуляция:

  • Валидация данных
  • Скрытие реализации
  • Защита от некорректных изменений
  • Легче рефакторинг

4. Наследование

Повторное использование кода через иерархию

Наследование — extends

// ── Родительский класс ─────────────
public class Animal {
    protected String name;
    protected int    energy;

    public Animal(String name, int energy) {
        this.name   = name;
        this.energy = energy;
    }

    public void eat() {
        energy += 20;
        System.out.println(name +
          " ест [энергия: " + energy + "]");
    }

    public void sleep() {
        energy += 30;
        System.out.println(name +
          " спит [энергия: " + energy + "]");
    }

    public String getInfo() {
        return name + " (энергия: "+energy+")";
    }
}

// ── Дочерние классы ────────────────
public class Dog extends Animal {
    private String breed;

    public Dog(String name, String breed) {
        super(name, 100); // вызов родителя!
        this.breed = breed;
    }

    public void fetch() {   // свой метод
        energy -= 15;
        System.out.println(name +
          " приносит мяч 🎾");
    }

    @Override               // переопределение
    public String getInfo() {
        return super.getInfo() +
          " порода: " + breed;
    }
}
public class Cat extends Animal {
    private boolean isIndoor;

    public Cat(String name,
               boolean isIndoor) {
        super(name, 70);
        this.isIndoor = isIndoor;
    }

    public void purr() {
        System.out.println(name +
          " мурчит 😸");
    }
}

// ── Использование ──────────────────
Dog dog = new Dog("Рекс", "Лабрадор");
Cat cat = new Cat("Барсик", true);

dog.eat();    // из Animal
dog.sleep();  // из Animal
dog.fetch();  // свой метод Dog

cat.eat();    // из Animal
cat.purr();   // свой метод Cat

System.out.println(dog.getInfo());
// Рекс (энергия: 135) порода: Лабрадор
Ключевые слова

extends — наследование
super() — конструктор родителя
super.method() — метод родителя
@Override — переопределение

@Override — переопределение методов

public class Shape {
    protected String color;

    public Shape(String color) {
        this.color = color;
    }

    public double area() {
        return 0; // базовая реализация
    }

    public String describe() {
        return color + " фигура, " +
               "площадь: " + area();
    }
}

public class Circle extends Shape {
    private double radius;

    public Circle(String color,
                  double radius) {
        super(color);
        this.radius = radius;
    }

    @Override
    public double area() {
        return Math.PI * radius * radius;
    }
}

public class Rectangle extends Shape {
    private double w, h;

    public Rectangle(String c,
                     double w, double h) {
        super(c);
        this.w = w; this.h = h;
    }

    @Override
    public double area() {
        return w * h;
    }
}
// ── Полиморфизм! ───────────────────
Shape[] shapes = {
    new Circle("красный", 5),
    new Rectangle("синий", 4, 6),
    new Circle("зелёный", 3)
};

for (Shape s : shapes) {
    System.out.println(s.describe());
}
// красный фигура, площадь: 78.54
// синий фигура, площадь: 24.00
// зелёный фигура, площадь: 28.27
⚠️ Правила @Override
Сигнатура должна совпадать точно.
Нельзя сузить модификатор доступа.
Нельзя бросать новые checked исключения.
Зачем @Override

Компилятор проверит что метод действительно существует в родителе.

final — запрет изменений

// final класс — нельзя наследовать
public final class ImmutablePoint {
    private final double x;
    private final double y;

    public ImmutablePoint(double x,
                          double y) {
        this.x = x;
        this.y = y;
    }

    public double getX() { return x; }
    public double getY() { return y; }

    public ImmutablePoint translate(
            double dx, double dy) {
        // создаём НОВЫЙ объект
        return new ImmutablePoint(
                x + dx, y + dy);
    }
}
// class ExtendedPoint
//   extends ImmutablePoint {} // ❌ ERROR
public class Account {
    // final поле — задаётся один раз
    private final String accountId;
    private double balance;

    public Account(String id) {
        this.accountId = id; // один раз!
        // accountId = "other"; // ❌ ERROR
    }

    // final метод — нельзя переопределить
    public final String getId() {
        return accountId;
    }
}

📌 Итоги лекции

  • Класс — чертёж. Объект — экземпляр.
  • Поля хранят состояние, методы — поведение.
  • Инкапсуляция: поля private, доступ через методы.
  • Конструктор = имя класса. this() — цепочка.
  • Наследование через extends.
  • super() — вызов родительского конструктора.
  • @Override — переопределение метода.
  • final — запрет наследования/переопределения/изменения.
